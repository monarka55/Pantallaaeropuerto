        // RENDERIZAR TABLAS DIVIDIDAS Y ORDENADAS POR ESTADO Y HORA
        function renderFlights() {
            const searchVal = document.getElementById('search-input').value.toLowerCase().trim();

            renderTable('departures-table-body', flightsData.departures, searchVal, 'count-departures');
            renderTable('arrivals-table-body', flightsData.arrivals, searchVal, 'count-arrivals');

        function renderTable(tbodyId, data, searchVal, countId) {
            const tbody = document.getElementById(tbodyId);
            tbody.innerHTML = '';

            // Filtrar y ordenar dinámicamente por prioridad de estado
            const sortedList = [...data]
                .filter(item => {
                    return item.flight.toLowerCase().includes(searchVal) ||
                           item.location.toLowerCase().includes(searchVal) ||
                           item.airline.toLowerCase().includes(searchVal) ||
                           item.gate.toLowerCase().includes(searchVal);
                })
                .sort((a, b) => {
                    const prioA = statusPriority[a.status] || 99;
                    const prioB = statusPriority[b.status] || 99;
                    if (prioA !== prioB) return prioA - prioB;
                    return a.time.localeCompare(b.time); // Si tienen el mismo estado, ordenar por hora
                });

            document.getElementById(countId).innerText = `${sortedList.length} VUELOS`;

            sortedList.forEach((item) => {
                let statusClass = '';
                if (item.status === 'ON TIME') statusClass = 'status-on-time';
                else if (item.status === 'BOARDING') statusClass = 'status-boarding';
                else if (item.status === 'DELAYED') statusClass = 'status-delayed';
                else if (item.status === 'LANDED') statusClass = 'status-landed';
                else if (item.status === 'CANCELLED') statusClass = 'status-cancelled';

                const tr = document.createElement('tr');
                tr.className = 'fids-row text-slate-200 border-b border-slate-800/40';
                tr.innerHTML = `
                    <td class="py-3 px-3 font-bold text-amber-400">${item.time}</td>
                    <td class="py-3 px-3 font-bold text-slate-100">${item.flight}</td>
                    <td class="py-3 px-3 text-slate-200 font-semibold">${item.location}</td>
                    <td class="py-3 px-3 text-center font-bold text-amber-300">${item.gate}</td>

        // CHIME DE AEROPUERTO SINTETIZADO (WEB AUDIO API)
        function playChime() {
            return new Promise((resolve) => {
                if (!audioCtx) {
                    const AudioContextClass = window.AudioContext || window.webkitAudioContext;
                    audioCtx = new AudioContextClass();
                }
                if (audioCtx.state === 'suspended') {
                    audioCtx.resume();
                }

                const notes = [
                    { freq: 523.25, time: 0.0, duration: 0.5 },
                    { freq: 659.25, time: 0.25, duration: 0.5 },
                    { freq: 783.99, time: 0.50, duration: 0.5 },
                    { freq: 1046.50, time: 0.75, duration: 1.0 }
                ];

                const now = audioCtx.currentTime;

                notes.forEach((note) => {
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();

                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(note.freq, now + note.time);

                    gain.gain.setValueAtTime(0, now + note.time);
                    gain.gain.linearRampToValueAtTime(0.3, now + note.time + 0.04);
                    gain.gain.exponentialRampToValueAtTime(0.0001, now + note.time + note.duration);

                    osc.connect(gain);
                    gain.connect(audioCtx.destination);

                    osc.start(now + note.time);
                    osc.stop(now + note.time + note.duration);
                });

                setTimeout(resolve, 1600);
            });
        }

        // LOCUCIÓN DE ANUNCIO DE VUELO (WEB SPEECH API)
        async function announceFlight(flight, location, gate, status) {
            await playChime();

            let message = "";
            const flightClean = flight.split('').join(' ');

            if (status === 'BOARDING') {
                message = `Atención pasajeros. El vuelo ${flightClean} con destino a ${location} se encuentra en proceso de abordaje por la puerta ${gate}.`;
            } else if (status === 'DELAYED') {
                message = `Atención por favor. El vuelo ${flightClean} con destino a ${location} se encuentra retrasado. Informaremos oportunamente.`;
            } else if (status === 'LANDED') {
                message = `Anuncio de llegada. El vuelo ${flightClean} procedente de ${location} ha aterrizado. Pasajeros dirigirse a la puerta ${gate}.`;
            } else {
                message = `Atención pasajeros del vuelo ${flightClean} con destino a ${location}, por favor estén atentos a la puerta de embarque ${gate}.`;
            }

            if ('speechSynthesis' in window) {
                window.speechSynthesis.cancel();
                const utterance = new SpeechSynthesisUtterance(message);
                utterance.lang = 'es-ES';
                utterance.rate = 0.9;
                window.speechSynthesis.speak(utterance);
            }
        }

        // RELOJ EN TIEMPO REAL
        function updateClock() {
            const now = new Date();
            const h = String(now.getHours()).padStart(2, '0');
            const m = String(now.getMinutes()).padStart(2, '0');
            const s = String(now.getSeconds()).padStart(2, '0');
            document.getElementById('clock-display').innerText = `${h}:${m}:${s}`;

            const options = { weekday: 'short', month: 'short', day: 'numeric', year: 'numeric' };
            document.getElementById('date-display').innerText = now.toLocaleDateString('es-ES', options).toUpperCase();
        }
